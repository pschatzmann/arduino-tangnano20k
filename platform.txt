# Path to the RISC-V toolchain: the Zephyr SDK's multilib
# riscv64-zephyr-elf-gcc, which supports the rv32i2p0/ilp32 target our SoC
# needs. Installed automatically as a Boards Manager tool dependency (see
# package_nanotang_index.json, docs/RELEASING.md) - {runtime.tools.*.path}
# resolves to wherever arduino-cli unpacked it. Override compiler.path
# and/or compiler.prefix in a platform.local.txt next to this file if you'd
# rather point at a different toolchain (any bare-metal rv32i gcc works,
# e.g. riscv32-unknown-elf-).
compiler.path={runtime.tools.riscv-zephyr-elf-0.17.0.path}/bin/
compiler.prefix=riscv64-zephyr-elf-

# FPGA tools (yosys, nextpnr-himbaechel, apicula's gowin_pack,
# openFPGALoader): a trimmed YosysHQ oss-cad-suite build, also installed
# as a Boards Manager tool dependency and pinned to the version the
# gateware was tested with (see tools/package/make_fpga_tools.py). Passed
# to tools/build_bitstream.py and tools/upload.py, which search its bin/

fpga_tools.path={runtime.tools.oss-cad-suite-gowin-2026.9.25.path}

# rv32i plus "m" (Tools > Hardware Multiply/Divide) and "c" (Tools >
# Compressed Instructions), each driven by its menu in lockstep with the
# gateware - see boards.txt. The _zicsr_zifencei suffix only spells out

build.march_m=m

compiler.mabi={build.mabi}

# Overridden per-selection by the Tools > SD Card board menu (see
# boards.txt). Unlike the AI Accelerator flag above, this isn't about
# gateware size - libraries/SD is GPLv3 (see docs/LICENSING.md), and
# compiling it into a sketch makes that sketch's binary a GPLv3
# derivative. Disabled by default so that happens only when you
# explicitly opt in, not just by typing #include <SD.h>.
build.sd_defines=

build.defines=-DF_CPU={build.f_cpu} -DARDUINO=10819 -DARDUINO_{build.board} -DARDUINO_ARCH_TANGNANO20K -DTANGNANO20K_SPI_COUNT={build.spi_count} -DTANGNANO20K_I2C_COUNT={build.i2c_count} -DTANGNANO20K_PWM_AUDIO={build.pwm_audio} -DTANGNANO20K_AI_ACCEL={build.ai_accel} {build.sd_defines}

# Overridden per-selection by the Tools > Optimize board menu (see
# boards.txt); this is the fallback if a sketch/IDE somehow doesn't set it.
build.optimization_flags=-Os

# Overridden per-selection by the Tools > AI Accelerator board menu (see
# boards.txt): "0" or "1", passed to tools/build_bitstream.py to decide
# whether to synthesize the AI accelerator gateware at all - it isn't
# free like an unused software library, since everything in top.v gets
# synthesized into the bitstream regardless of what a sketch calls. Also
# exported to sketches as TANGNANO20K_AI_ACCEL (see build.defines above).
build.ai_accel=0

# Overridden by the Tools > Boot Mode board menu (see boards.txt). "0"
# (SRAM, default) keeps the original always-resynthesize model; "1"
# (Flash) bakes only the fixed boot stub into SRAM and writes the sketch
# itself to the onboard flash instead - see tools/build_bitstream.py and
# docs/PERIPHERALS.md "Flash".
build.boot_flash=0

# Overridden by the Tools > SPI Buses / I2C Buses board menus (see
# boards.txt): "1" (default, each) is the original always-present primary
# port; "0" removes it entirely to save LUTs; "2" adds a second,
# independent port (SPI2 on GPIO0-3 / Wire2 on GPIO4-5) - see
# tools/build_bitstream.py and docs/PERIPHERALS.md.
build.spi_count=1
build.i2c_count=1

# Overridden by the Tools > Hardware Multiply/Divide board menu (see
# boards.txt), which also overrides build.march_m and build.libgcc_path in
# lockstep - see that menu's comment there for why they must move
# together. "1" (default) = real M-extension hardware; "0" = software
# mul/div/float.
build.hw_muldiv=1

# Overridden by the Tools > Compressed Instructions / Barrel Shifter board
# menus (see boards.txt): "0" (default) = off, matching every previous
# build of this core.
build.compressed=0
build.barrel_shifter=0

# Overridden by the Tools > CAN board menu (see boards.txt): "0" (default)
# leaves the CAN controller out of the gateware.
build.can=0

# Overridden by the Tools > Boot Mode board menu (see boards.txt):
# link_cmd.ld (default) keeps everything in the internal SRAM;
# link_cmd_sdram.ld (the "+ SDRAM" entries) moves code off the hot
# path into SDRAM.
build.ldscript=link_cmd.ld

# Overridden by the Tools > I2S Input board menu (see boards.txt): "0"
# (default) leaves GPIO6 as plain GPIO and I2S.read() always reads back
# zero; "1" wires GPIO6 to the I2S peripheral's receive input for an
# external I2S microphone - see tools/build_bitstream.py and
# docs/PERIPHERALS.md "Audio (I2S)".
build.i2s_rx=0

# Overridden by the Tools > PWM Audio board menu (see boards.txt): "0"
# (default) leaves GPIO16/GPIO17 as plain GPIO; "1" synthesizes the PWM
# audio peripheral onto them - see tools/build_bitstream.py and
# docs/PERIPHERALS.md "Audio (PWM)". Also exported to sketches as
# TANGNANO20K_PWM_AUDIO (see build.defines above).
build.pwm_audio=0

# Overridden by the Tools > Flash Cache board menu (see boards.txt): "0"
# (default) keeps the plain flash reader; "1" synthesizes the 512-byte
# read cache in front of it (gateware/src/qspi_flash_cached.v) - see
# tools/build_bitstream.py and docs/PERIPHERALS.md "Flash".
build.flash_cache=0

# Overridden by the Tools > Clock Speed board menu (see boards.txt), which
# also overrides build.f_cpu in lockstep - all four move together, same
# pattern as Hardware Multiply/Divide's build.march/build.libgcc_path,
# since clk_freq_hz/pll_idiv/pll_fbdiv/pll_odiv describe one single PLL
# configuration and build.f_cpu must match what it actually produces (see
# tools/build_bitstream.py). Defaults here are Normal (27MHz) - the
# original fixed clock this core shipped with.
build.f_cpu=27000000L
build.clk_freq_hz=27000000
build.pll_idiv=0
build.pll_fbdiv=0
build.pll_odiv=32

compiler.c.cmd={compiler.prefix}gcc
compiler.c.flags=-c {build.optimization_flags} -ffunction-sections -fdata-sections -march={compiler.march} -mabi={compiler.mabi} -nostdlib -nostartfiles -std=gnu11 {compiler.warning_flags} {build.defines}

compiler.c.elf.cmd={compiler.prefix}g++

# -Wl,--undefined=irq_vec and =_boot: irq_vec.S/boot.S are only ever
# reached via picorv32's hardware PROGADDR_IRQ/PROGADDR_RESET jumps, never
# referenced by any C symbol - so without this, the linker's normal
# reference-driven archive extraction (core.a is linked without
# --whole-archive, deliberately, to avoid pulling in unused peripheral
# code - see the recipe.c.combine.pattern comment history) never even
# pulls their object files out of core.a in the first place, silently
# leaving the actual interrupt handler and boot stub code entirely absent
# from the linked binary. This was a real, previously-undetected bug this
# project had - see docs/KNOWN_LIMITATIONS.md.
compiler.c.elf.flags=-nostdlib -nostartfiles -march={compiler.march} -mabi={compiler.mabi} {build.optimization_flags} -Wl,--gc-sections -Wl,--undefined=irq_vec -Wl,--undefined=_boot -T{build.core.path}/{build.ldscript} {build.exceptions_ldflags} -Wl,-Map={build.path}/{build.project_name}.map

compiler.cpp.cmd={compiler.prefix}g++
# Overridden per-selection by the Tools > C++ Exceptions board menu (see
# boards.txt), disabled by default.
build.exceptions_flags=-fno-exceptions
# Selects which eh_frame.ld the linker scripts INCLUDE (-L search path):
# cores/tangnano20k/ldscripts/exceptions/ keeps the unwind tables,
# ldscripts/no_exceptions/ discards them.
build.exceptions_ldflags="-L{build.core.path}/ldscripts/no_exceptions"

compiler.cpp.flags=-c {build.optimization_flags} -ffunction-sections -fdata-sections -march={compiler.march} -mabi={compiler.mabi} -nostdlib -nostartfiles {build.exceptions_flags} -fno-rtti -fno-threadsafe-statics -std=gnu++17 {compiler.warning_flags} {build.defines}

compiler.S.cmd={compiler.prefix}gcc
compiler.S.flags=-c -march={compiler.march} -mabi={compiler.mabi} -x assembler-with-cpp

compiler.ar.cmd={compiler.prefix}ar
compiler.ar.flags=rcs

compiler.objcopy.cmd={compiler.prefix}objcopy
compiler.size.cmd={compiler.prefix}size

# build.libgcc_path is set per Tools > Hardware Multiply/Divide selection
# (see boards.txt) - each menu option's exact path was confirmed via `gcc
# -print-libgcc-file-name` for that -march, not resolved at build time
# (arduino-cli's recipe $(...) shell-out didn't play well with nested
# property substitution here). Update both menu options in boards.txt if
# you point compiler.path at a different toolchain / gcc version in
# platform.local.txt.
compiler.libgcc={build.libgcc_path}

compiler.libraries.ldflags=

# The vendored ArduinoCore-API headers live at cores/tangnano20k/api/ (one
# level below the core), and third-party libraries (e.g. libraries/SD)
# commonly do `#include <Print.h>`/`<Stream.h>` etc. expecting them on a
# plain include path, not a subdirectory - so this needs to reach every
# compile, not just the core's own sources.
build.extra_flags=-I{build.core.path}/api

# ---------------------------------------------------------------------------
# Compile recipes
# ---------------------------------------------------------------------------
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} "{archive_file_path}" "{object_file}"

# No --whole-archive: core.a's members should only be pulled in when
# actually referenced, so a sketch that doesn't touch e.g. Serial doesn't
# pay for HardwareSerial.cpp's code - important given the 64KB SRAM budget.
#
# The toolchain's newlib libm/libc and libstdc++ are linked too, for what
# the core doesn't implement itself (<math.h>, the rest of <ctype.h>,
# std::string, ...). -nostdlib drops them from the default link but not
# the -march-matching multilib search path, so plain -l finds the right
# ones. core.a comes first inside the group, so its own malloc(), printf(),
# mem*()/str*() and abort() always win over newlib's - libc only supplies
# symbols nothing earlier defines.
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} -o {build.path}/{build.project_name}.elf {object_files} -Wl,--start-group "{archive_file_path}" -lstdc++ -lm -lc {compiler.libgcc} -Wl,--end-group

# Repurposed: instead of an objcopy-to-hex step, this drives the full FPGA
# build (bake the program into SRAM init files, then yosys -> nextpnr ->
# gowin_pack). Expect this to take noticeably longer than a normal
# Arduino "compile" step - see docs/BUILDING.md.
recipe.objcopy.hex.pattern=python3 "{runtime.platform.path}/tools/build_bitstream.py" "--tools-dir={fpga_tools.path}" "{build.path}/{build.project_name}.elf" "{compiler.path}{compiler.objcopy.cmd}" "{build.path}/fpga" "{build.ai_accel}" "{build.boot_flash}" "{build.spi_count}" "{build.i2c_count}" "{build.hw_muldiv}" "{build.i2s_rx}" "{build.clk_freq_hz}" "{build.pll_idiv}" "{build.pll_fbdiv}" "{build.pll_odiv}" "{build.pwm_audio}" "{build.flash_cache}" "{build.compressed}" "{build.barrel_shifter}" "{build.can}"

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
# Everything except .flash_data lives in the one 64KB internal SRAM (see
# cores/tangnano20k/link_cmd.ld): code, constants, variables and the IRQ
# scratch area share it. So "program storage" is the whole SRAM footprint
# (compared against upload.maximum_size), and "global variables" is the
# .data/.bss part of it. .flash_data (Boot Mode: Flash) is not counted,
# and neither is .sdram_image (Tools > Boot Mode: ... + SDRAM).
recipe.size.regex=^(?:\.text\S*|\.irq_scratch|\.rodata\S*|\.init_array|\.eh_frame|\.gcc_except_table|\.data|\.bss)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss)\s+([0-9]+).*
